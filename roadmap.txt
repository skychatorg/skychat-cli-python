from textual.app import App, ComposeResult
from textual.widgets import Header, Footer, Static
from textual_image import Image as TextualImage  # ← this one auto-detects
from PIL import Image as PilImage

class AutoImageApp(App):
    def compose(self) -> ComposeResult:
        yield Header()
        yield Footer()

        pil_img = PilImage.open("example.jpg").convert("RGB")
        pil_img.thumbnail((500, 400))  # recommended: keep reasonable size

        # This will try:
        # 1. Kitty graphics protocol (TGP) if detected
        # 2. Sixel if available (works great in foot)
        # 3. Unicode half-cell / block fallback if neither
        yield Static(TextualImage(pil_img))

if __name__ == "__main__":
    AutoImageApp().run()

from __future__ import annotations
from pathlib import Path
import tempfile

from textual.app import App, ComposeResult, on
from textual.widgets import Header, Footer, Static, Input, Button
from textual.containers import Vertical
from textual.events import Paste
from textual_image import Image as TextualImage  # pip install textual-image
from PIL import Image as PilImage


class ImagePasteDemo(App):
    CSS = """
    Screen { background: $surface; }
    #preview { height: 60%; width: 100%; align: center middle; }
    #status  { height: 2; content-align: center middle; }
    """

    def compose(self) -> ComposeResult:
        yield Header()
        yield Vertical(
            Static(id="preview"),
            Input(placeholder="Paste image path here or drag & drop…", id="path"),
            Button("Clear", id="clear"),
            id="main"
        )
        yield Footer()
        yield Static("Paste a file path or drop an image file", id="status")

    @on(Paste)
    def on_paste(self, event: Paste) -> None:
        text = event.text.strip()
        if not text:
            return

        # Often drag & drop pastes file:// or just /path/to/file.jpg
        if text.startswith("file://"):
            text = text[7:]
        path = Path(text).expanduser()

        if path.is_file() and path.suffix.lower() in {".png", ".jpg", ".jpeg", ".gif", ".webp"}:
            self.display_image(path)
            self.query_one("#status").update(f"Loaded: {path.name}")
        else:
            self.query_one("#status").update("Not a valid image path")

    def display_image(self, path: Path | str) -> None:
        try:
            pil_img = PilImage.open(path).convert("RGB")
            pil_img.thumbnail((800, 600))  # sane size for terminal

            preview = self.query_one("#preview", Static)
            preview.update(TextualImage(pil_img))  # auto: sixel in foot, TGP in kitty
        except Exception as e:
            self.query_one("#status").update(f"Error: {e}")

    @on(Button.Pressed, "#clear")
    def clear_image(self):
        self.query_one("#preview", Static).update("")
        self.query_one("#status").update("Cleared")


if __name__ == "__main__":
    ImagePasteDemo().run()
